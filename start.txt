# Set up and work in a virtual environment
python3 -m venv VENV
bash ./VENV/bin/activate

# Install initial dependencies
pip uninstall discord.py
pip install py-cord
pip install "py-cord[voice]"
pip install python-dotenv
pip install gtts
pip install python-dateutil
pip install youtube-dl
pip3 install ffmpeg-normalize

# you can try making a directory in this bot called youtube_dl,

# Invite the bot to your server(s)
# Do not invite the bot with the token TEST_MODE_BOT_TOKEN to non-test servers!
# If you don't want or can find a way around any of these permissions, feel free
# to disable them, the bot just might not work as intended. The less
# permissions needed, the better!
# In OAuth2 tab:
# - Scopes:
#   - bot (to let users see this 'Discord user' is a bot)
# - Bot Permissions:
#   - General Permissions:
#     - Read Messages/View Channels (to let bot read polls it created)
#   - Text Permissions:
#     - Send Messages (to let bot send reminders/polls, not just responses)
#     - Send TTS Messages (to let users use TTS on bot's text responses)
#     - Send Messages in Threads (to let bot respond to in-thread commands)
#     - Embed Links (to let bot give links with pictures)
#     - Attach Files (to let bot send audio from its sound library)
#     - Read Message History (to let bot read polls it created)
#     - Add Reactions (to let bot react to polls it created)
#   - Voice Permissions:
#     - Connect (to let bot join a voice channel)
#     - Speak (to let bot be able to play audio in a voice channel)
#     - Use Voice Activity (to let bot be able to set voice status)
# Result: https://discord.com/api/oauth2/authorize?client_id=???&permissions=274914737216&scope=bot
